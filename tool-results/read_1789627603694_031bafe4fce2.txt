    93→function QrPanel({
    94→  sessions,
    95→  activeSession,
    96→  onSelect,
    97→  onCreate,
    98→  onUpdate,
    99→  onDelete,
   100→  creating,
   101→}: {
   102→  sessions: SessionShape[];
   103→  activeSession: SessionShape | null;
   104→  onSelect: (id: string) => void;
   105→  onCreate: (name: string) => void;
   106→  onUpdate: (id: string, patch: { closed?: boolean; name?: string }) => void;
   107→  onDelete: (id: string) => void;
   108→  creating: boolean;
   109→}) {
   110→  const [copied, setCopied] = useState<"link" | "code" | null>(null);
   111→  const [newName, setNewName] = useState("");
   112→  const [showNew, setShowNew] = useState(false);
   113→  const [confirmDelete, setConfirmDelete] = useState(false);
   114→
   115→  const uploadUrl = useMemo(() => {
   116→    if (!activeSession) return "";
   117→    if (typeof window === "undefined") return `/?upload=${activeSession.code}`;
   118→    return `${window.location.origin}/?upload=${activeSession.code}`;
   119→  }, [activeSession]);
   120→
   121→  const copy = async (text: string, kind: "link" | "code") => {
   122→    try {
   123→      await navigator.clipboard.writeText(text);
   124→      setCopied(kind);
   125→      setTimeout(() => setCopied(null), 1600);
   126→    } catch {
   127→      /* ignore */
   128→    }
   129→  };
   130→
   131→  return (
   132→    <Card className="lg:sticky lg:top-6 flex flex-col gap-0 overflow-hidden">
   133→      <CardHeader className="pb-3">
   134→        <CardTitle className="flex items-center gap-2 text-base">
   135→          <QrCode className="h-4 w-4 text-emerald-600" />
   136→          Collection QR
   137→        </CardTitle>
   138→        <CardDescription>
   139→          Show this to students. They scan &amp; upload.
   140→        </CardDescription>
   141→      </CardHeader>
   142→      <CardContent className="flex flex-col gap-4">
   143→        {/* Session selector */}
   144→        <div className="space-y-2">
   145→          <Label className="text-xs text-muted-foreground">Active session</Label>
   146→          <div className="flex gap-2">
   147→            <Select
   148→              value={activeSession?.id ?? ""}
   149→              onValueChange={onSelect}
   150→              disabled={sessions.length === 0}
   151→            >
   152→              <SelectTrigger className="flex-1">
   153→                <SelectValue
   154→                  placeholder={
   155→                    sessions.length === 0
   156→                      ? "No sessions yet"
   157→                      : "Select session"
   158→                  }
   159→                />
   160→              </SelectTrigger>
   161→              <SelectContent>
   162→                {sessions.map((s) => (
   163→                  <SelectItem key={s.id} value={s.id}>
   164→                    <span className="flex items-center gap-2">
   165→                      {s.closed && (
   166→                        <Lock className="h-3 w-3 text-muted-foreground" />
   167→                      )}
   168→                      <span className="truncate">{s.name}</span>
   169→                      <Badge
   170→                        variant="secondary"
   171→                        className="ml-1 h-4 px-1 text-[10px]"
   172→                      >
   173→                        {s.code}
   174→                      </Badge>
   175→                    </span>
   176→                  </SelectItem>
   177→                ))}
   178→              </SelectContent>
   179→            </Select>
   180→            <Button
   181→              size="icon"
   182→              variant="outline"
   183→              onClick={() => setShowNew(true)}
   184→              title="New session"
   185→            >
   186→              <Plus className="h-4 w-4" />
   187→            </Button>
   188→          </div>
   189→        </div>
   190→
   191→        {/* QR display */}
   192→        {activeSession ? (
   193→          <div className="flex flex-col items-center gap-3 rounded-xl border bg-gradient-to-b from-emerald-50/60 to-transparent p-4">
   194→            <div className="rounded-xl bg-white p-3 shadow-sm ring-1 ring-black/5">
   195→              <QRCodeSVG
   196→                value={uploadUrl}
   197→                size={208}
   198→                level="M"
   199→                marginSize={1}
   200→                fgColor="#0f172a"
   201→              />
   202→            </div>
   203→            <div className="flex w-full items-center justify-between rounded-lg bg-white px-3 py-2 ring-1 ring-black/5">
   204→              <div className="min-w-0">
   205→                <p className="text-[11px] uppercase tracking-wide text-muted-foreground">
   206→                  Session code
   207→                </p>
   208→                <p className="font-mono text-lg font-bold tracking-widest text-slate-900">
   209→                  {activeSession.code}
   210→                </p>
   211→              </div>
   212→              <Button
   213→                size="sm"
   214→                variant="ghost"
   215→                onClick={() => copy(activeSession.code, "code")}
   216→              >
   217→                {copied === "code" ? (
   218→                  <Check className="h-4 w-4 text-emerald-600" />
   219→                ) : (
   220→                  <Copy className="h-4 w-4" />
   221→                )}
   222→              </Button>
   223→            </div>
   224→
   225→            <div className="flex w-full flex-col gap-2">
   226→              <Button
   227→                variant="outline"
   228→                className="w-full justify-start"
   229→                onClick={() => copy(uploadUrl, "link")}
   230→              >
   231→                {copied === "link" ? (
   232→                  <Check className="h-4 w-4 text-emerald-600" />
   233→                ) : (
   234→                  <Copy className="h-4 w-4" />
   235→                )}
   236→                Copy upload link
   237→              </Button>
   238→              <Button
   239→                variant="outline"
   240→                className="w-full justify-start"
   241→                onClick={() => window.open(uploadUrl, "_blank")}
   242→              >
   243→                <ExternalLink className="h-4 w-4" />
   244→                Open student page
   245→              </Button>
   246→            </div>
   247→
   248→            <Separator />
   249→
   250→            <div className="flex w-full items-center justify-between">
   251→              <div className="flex items-center gap-2">
   252→                <Label
   253→                  htmlFor="closed-toggle"
   254→                  className="text-sm text-muted-foreground"
   255→                >
   256→                  {activeSession.closed ? "Closed" : "Accepting uploads"}
   257→                </Label>
   258→              </div>
   259→              <Switch
   260→                id="closed-toggle"
   261→                checked={!activeSession.closed}
   262→                onCheckedChange={(checked) =>
   263→                  onUpdate(activeSession.id, { closed: !checked })
   264→                }
   265→              />
   266→            </div>
   267→
   268→            <Button
   269→              variant="ghost"
   270→              size="sm"
   271→              className="w-full text-destructive hover:text-destructive"
   272→              onClick={() => setConfirmDelete(true)}
   273→            >
   274→              <Trash2 className="h-3.5 w-3.5" />
   275→              Delete session &amp; all files
   276→            </Button>
   277→          </div>
   278→        ) : (
   279→          <div className="flex flex-col items-center gap-3 rounded-xl border border-dashed p-8 text-center">
   280→            <FolderOpen className="h-10 w-10 text-muted-foreground/50" />
   281→            <p className="text-sm text-muted-foreground">
   282→              Create your first session to generate a QR code.
   283→            </p>
   284→            <Button onClick={() => setShowNew(true)}>
   285→              <Plus className="h-4 w-4" />
   286→              New session
   287→            </Button>
   288→          </div>
   289→        )}
   290→      </CardContent>
   291→
   292→      {/* New session dialog */}
   293→      <Dialog open={showNew} onOpenChange={setShowNew}>
   294→        <DialogContent>
   295→          <DialogHeader>
   296→            <DialogTitle>New collection session</DialogTitle>

   300→          </DialogHeader>
   301→          <div className="py-2">

   305→              value={newName}
   306→              onChange={(e) => setNewName(e.target.value)}

   311→                  setShowNew(false);
   312→                }
   313→              }}
   314→            />
   315→          </div>
   316→          <DialogFooter>
   317→            <Button
   318→              variant="outline"
   319→              onClick={() => setShowNew(false)}
   320→            >
   321→              Cancel
   322→            </Button>
   323→            <Button
   324→              disabled={!newName.trim() || creating}
   325→              onClick={() => {
   326→                onCreate(newName.trim());
   327→                setNewName("");
   328→                setShowNew(false);
   329→              }}
   330→            >
   331→              {creating && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   332→              Create
   333→            </Button>
   334→          </DialogFooter>
   335→        </DialogContent>
   336→      </Dialog>
   337→
   338→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   339→        <AlertDialogContent>
   340→          <AlertDialogHeader>
   341→            <AlertDialogTitle>Delete this session?</AlertDialogTitle>
   342→            <AlertDialogDescription>
   343→              This permanently deletes <b>{activeSession?.name}</b> and{" "}
   344→              {activeSession?.fileCount ?? 0} file(s). This cannot be undone.
   345→            </AlertDialogDescription>
   346→          </AlertDialogHeader>
   347→          <AlertDialogFooter>
   348→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   349→            <AlertDialogAction
   350→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   351→              onClick={() => {
   352→                if (activeSession) onDelete(activeSession.id);
   353→                setConfirmDelete(false);
   354→              }}
   355→            >
   356→              Delete
   357→            </AlertDialogAction>
   358→          </AlertDialogFooter>
   359→        </AlertDialogContent>
   360→      </AlertDialog>
   361→    </Card>
   362→  );
   363→}

   542→/*  Settings dialog                                                   */
   543→/* ------------------------------------------------------------------ */
   544→
   545→function SettingsDialog({
   546→  open,
   547→  onOpenChange,
   548→  settings,
   549→  onSave,
   550→}: {
   551→  open: boolean;
   552→  onOpenChange: (o: boolean) => void;
   553→  settings: SettingsShape;
   554→  onSave: (s: Partial<SettingsShape>) => Promise<void>;
   555→}) {
   556→  const [autoDelete, setAutoDelete] = useState(settings.autoDeleteAfterPrint);
   557→  const [retention, setRetention] = useState(settings.retentionHours);
   558→  const [saving, setSaving] = useState(false);
   559→
   560→  useEffect(() => {
   561→    setAutoDelete(settings.autoDeleteAfterPrint);
   562→    setRetention(settings.retentionHours);
   563→  }, [settings]);
   564→
   565→  const save = async () => {
   566→    setSaving(true);
   567→    try {
   568→      await onSave({
   569→        autoDeleteAfterPrint: autoDelete,
   570→        retentionHours: Number(retention),
   571→      });
   572→      onOpenChange(false);
   573→    } finally {
   574→      setSaving(false);
   575→    }
   576→  };
   577→
   578→  return (
   579→    <Dialog open={open} onOpenChange={onOpenChange}>
   580→      <DialogContent>
   581→        <DialogHeader>
   582→          <DialogTitle className="flex items-center gap-2">
   583→            <SettingsIcon className="h-4 w-4" />
   584→            Auto-delete settings
   585→          </DialogTitle>
   586→          <DialogDescription>
   587→            Control how long uploaded files live before being removed.
   588→          </DialogDescription>
   589→        </DialogHeader>
   590→        <div className="space-y-5 py-2">
   591→          <div className="flex items-start justify-between gap-4 rounded-lg border p-3">
   592→            <div>
   593→              <Label className="text-sm font-medium">
   594→                Delete after printing
   595→              </Label>
   596→              <p className="mt-0.5 text-xs text-muted-foreground">
   597→                When you mark a file as printed, it is deleted automatically.
   598→              </p>
   599→            </div>
   600→            <Switch
   601→              checked={autoDelete}
   602→              onCheckedChange={setAutoDelete}
   603→            />
   604→          </div>
   605→          <div className="rounded-lg border p-3">
   606→            <Label className="text-sm font-medium">
   607→              Auto-expire uploads
   608→            </Label>
   609→            <p className="mt-0.5 text-xs text-muted-foreground">
   610→              Files are deleted this long after upload. Set to 0 to disable.
   611→            </p>
   612→            <div className="mt-3 flex items-center gap-2">
   613→              <Input
   614→                type="number"
   615→                min={0}
   616→                max={168}
   617→                value={retention}
   618→                onChange={(e) => setRetention(Number(e.target.value))}
   619→                className="w-24"
   620→              />
   621→              <span className="text-sm text-muted-foreground">hours</span>
   622→            </div>
   623→          </div>
   624→        </div>
   625→        <DialogFooter>
   626→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   627→            Cancel
   628→          </Button>
   629→          <Button onClick={save} disabled={saving}>
   630→            {saving && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   631→            Save
   632→          </Button>
   633→        </DialogFooter>
   634→      </DialogContent>
   635→    </Dialog>
   636→  );
   637→}

   649→    autoDeleteAfterPrint: false,
   650→    retentionHours: 1,
   651→  });
   652→  const [loadingFiles, setLoadingFiles] = useState(false);
   653→  const [creating, setCreating] = useState(false);
   654→  const [printingId, setPrintingId] = useState<string | null>(null);
   655→  const [deletingId, setDeletingId] = useState<string | null>(null);
   656→  const [viewFile, setViewFile] = useState<FileShape | null>(null);
   657→  const [settingsOpen, setSettingsOpen] = useState(false);
   658→  const [confirmDeleteFile, setConfirmDeleteFile] = useState<FileShape | null>(null);
   659→  const [confirmClearAll, setConfirmClearAll] = useState(false);
   660→  const [connected, setConnected] = useState(false);
   661→  const [cleaning, setCleaning] = useState(false);
   662→
   663→  const activeSession = useMemo(
   664→    () => sessions.find((s) => s.id === activeId) ?? null,
   665→    [sessions, activeId],
   666→  );
   667→
   668→  const refreshSessions = useCallback(async () => {
   669→    try {
   670→      const list = await listSessions();
   671→      setSessions(list);
   672→      // Auto-select the first open session (or first overall) if none selected.
   673→      setActiveId((prev) => {
   674→        if (prev && list.some((s) => s.id === prev)) return prev;
   675→        const open = list.find((s) => !s.closed);
   676→        return (open ?? list[0])?.id ?? null;
   677→      });
   678→    } catch (e) {
   679→      toast({
   680→        title: "Could not load sessions",
   681→        description: (e as Error).message,
   682→        variant: "destructive",
   683→      });
   684→    }
   685→  }, [toast]);
   686→
   687→  const refreshFiles = useCallback(async () => {
   688→    if (!activeId) {
   689→      setFiles([]);
   690→      return;
   691→    }
   692→    try {
   693→      const f = await listFiles(activeId);
   694→      setFiles(f);
   695→    } catch {
   696→      /* ignore transient */
   697→    } finally {
   698→      setLoadingFiles(false);
   699→    }
   700→  }, [activeId]);
   701→
   702→  const refreshSettings = useCallback(async () => {
   703→    try {
   704→      setSettings(await getSettings());
   705→    } catch {
   706→      /* ignore */
   707→    }
   708→  }, []);
   709→
   710→  // Initial load.
   711→  useEffect(() => {
   712→    refreshSessions();
   713→    refreshSettings();
   714→  }, [refreshSessions, refreshSettings]);
   715→
   716→  // Load files when session changes + poll every 4s as a reliable fallback.
   717→  useEffect(() => {
   718→    setLoadingFiles(true);
   719→    refreshFiles();
   720→    if (!activeId) return;
   721→    const t = setInterval(refreshFiles, 4000);
   722→    return () => clearInterval(t);
   723→  }, [activeId, refreshFiles]);
   724→
   725→  // Real-time socket. Handlers merge into the file list (and refresh session counts).
   726→  useAdminSocket(
   727→    {
   728→      "file-uploaded": (data) => {
   729→        setFiles((prev) => {
   730→          if (prev.some((f) => f.id === data.file.id)) return prev;
   731→          return [data.file, ...prev];
   732→        });
   733→        setSessions((prev) =>
   734→          prev.map((s) =>
   735→            s.id === data.file.sessionId
   736→              ? { ...s, fileCount: s.fileCount + 1 }
   737→              : s,
   738→          ),
   739→        );
   740→      },
   741→      "file-deleted": (data) => {
   742→        setFiles((prev) => prev.filter((f) => f.id !== data.fileId));
   743→      },
   744→      "file-printed": (data) => {
   745→        setFiles((prev) =>
   746→          prev.map((f) => (f.id === data.file.id ? data.file : f)),
   747→        );
   748→      },
   749→      "session-updated": (data) => {
   750→        setSessions((prev) =>
   751→          prev.map((s) => (s.id === data.session.id ? data.session : s)),
   752→        );
   753→      },
   754→    },
   755→    setConnected,
   756→  );
   757→
   758→  /* ---- Actions ---- */
   759→
   760→  const handleCreate = async (name: string) => {
   761→    setCreating(true);
   762→    try {
   763→      const s = await createSession(name);
   764→      setSessions((prev) => [s, ...prev]);
   765→      setActiveId(s.id);
   766→      toast({ title: "Session created", description: s.name });
   767→    } catch (e) {
   768→      toast({
   769→        title: "Failed to create session",
   770→        description: (e as Error).message,
   771→        variant: "destructive",
   772→      });
   773→    } finally {
   774→      setCreating(false);
   775→    }
   776→  };
   777→
   778→  const handleUpdateSession = async (
   779→    id: string,
   780→    patch: { closed?: boolean; name?: string },
   781→  ) => {
   782→    try {
   783→      const updated = await updateSession(id, patch);
   784→      setSessions((prev) => prev.map((s) => (s.id === id ? updated : s)));
   785→    } catch (e) {
   786→      toast({
   787→        title: "Failed to update session",
   788→        description: (e as Error).message,
   789→        variant: "destructive",
   790→      });
   791→    }
   792→  };
   793→
   794→  const handleDeleteSession = async (id: string) => {
   795→    try {
   796→      await deleteSession(id);
   797→      setSessions((prev) => prev.filter((s) => s.id !== id));
   798→      setFiles([]);
   799→      setActiveId(null);
   800→      toast({ title: "Session deleted" });
   801→    } catch (e) {
   802→      toast({
   803→        title: "Failed to delete session",
   804→        description: (e as Error).message,
   805→        variant: "destructive",
   806→      });
   807→    }
   808→  };
   809→
   810→  const handlePrint = async (file: FileShape) => {
   811→    setPrintingId(file.id);
   812→    // Open the print HTML in a new tab; it auto-triggers window.print().
   813→    window.open(printUrl(file.id), "_blank", "noopener,noreferrer");
   814→    // Mark as printed (this may auto-delete if the setting is on).
   815→    try {
   816→      const updated = await markPrinted(file.id, true);
   817→      if (updated) {
   818→        setFiles((prev) =>
   819→          prev.map((f) => (f.id === file.id ? updated : f)),
   820→        );
   821→      } else {
   822→        // auto-deleted after print
   823→        setFiles((prev) => prev.filter((f) => f.id !== file.id));
   824→        toast({
   825→          title: "Printed & auto-deleted",
   826→          description: file.filename,
   827→        });
   828→      }
   829→    } catch {
   830→      /* ignore — print window still opened */
   831→    } finally {
   832→      setPrintingId(null);
   833→    }
   834→  };
   835→
   836→  const handleDeleteFile = async (id: string) => {
   837→    setDeletingId(id);
   838→    try {
   839→      await deleteFile(id);
   840→      setFiles((prev) => prev.filter((f) => f.id !== id));
   841→      setSessions((prev) =>
   842→        prev.map((s) => {
   843→          const f = files.find((x) => x.id === id);
   844→          return f && s.id === f.sessionId
   845→            ? { ...s, fileCount: Math.max(0, s.fileCount - 1) }
   846→            : s;
   847→        }),
   848→      );
   849→      toast({ title: "File deleted" });
   850→    } catch (e) {
   851→      toast({
   852→        title: "Failed to delete file",
   853→        description: (e as Error).message,
   854→        variant: "destructive",
   855→      });
   856→    } finally {
   857→      setDeletingId(null);
   858→    }
   859→  };
   860→
   861→  const handleClearAll = async () => {
   862→    setConfirmClearAll(false);
   863→    const targets = [...files];
   864→    for (const f of targets) {
   865→      try {
   866→        await deleteFile(f.id);
   867→      } catch {
   868→        /* ignore */
   869→      }
   870→    }
   871→    setFiles([]);
   872→    setSessions((prev) =>
   873→      prev.map((s) =>
   874→        s.id === activeId ? { ...s, fileCount: 0 } : s,
   875→      ),
   876→    );
   877→    toast({ title: `Deleted ${targets.length} file(s)` });
   878→  };
   879→
   880→  const handleCleanup = async () => {
   881→    setCleaning(true);
   882→    try {
   883→      const n = await runCleanup();
   884→      await refreshFiles();
   885→      toast({ title: `Cleanup done`, description: `${n} expired file(s) removed` });
   886→    } catch (e) {
   887→      toast({
   888→        title: "Cleanup failed",
   889→        description: (e as Error).message,
   890→        variant: "destructive",
   891→      });
   892→    } finally {
   893→      setCleaning(false);
   894→    }
   895→  };
   896→
   897→  const handleSaveSettings = async (patch: Partial<SettingsShape>) => {
   898→    try {
   899→      const s = await updateSettings(patch);
   900→      setSettings(s);
   901→      toast({ title: "Settings saved" });
   902→    } catch (e) {
   903→      toast({
   904→        title: "Failed to save settings",
   905→        description: (e as Error).message,
   906→        variant: "destructive",
   907→      });
   908→    }
   909→  };
   910→
   911→  const totalSize = useMemo(
   912→    () => files.reduce((acc, f) => acc + f.size, 0),
   913→    [files],
   914→  );
   915→  const printedCount = files.filter((f) => f.printed).length;
   916→
   917→  return (
   918→    <div className="flex min-h-screen flex-col bg-muted/20">
   919→      {/* Header */}
   920→      <header className="sticky top-0 z-30 border-b bg-background/80 backdrop-blur">
   921→        <div className="mx-auto flex h-14 max-w-7xl items-center justify-between px-4">
   922→          <div className="flex items-center gap-2.5">
   923→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-600 text-white shadow-sm">
   924→              <QrCode className="h-4 w-4" />
   925→            </div>
   926→            <div className="leading-tight">
   927→              <h1 className="text-sm font-semibold">QR File Collector</h1>
   928→              <p className="text-[11px] text-muted-foreground">
   929→                Admin dashboard
   930→              </p>
   931→            </div>
   932→          </div>
   933→          <div className="flex items-center gap-1.5">
   934→            <Badge
   935→              variant="outline"
   936→              className={`hidden sm:flex items-center gap-1 ${
   937→                connected ? "border-emerald-300 text-emerald-700" : "text-muted-foreground"
   938→              }`}
   939→            >
   940→              {connected ? (
   941→                <Wifi className="h-3 w-3" />
   942→              ) : (
   943→                <WifiOff className="h-3 w-3" />
   944→              )}
   945→              {connected ? "Live" : "Polling"}
   946→            </Badge>
   947→            <Button
   948→              variant="ghost"
   949→              size="icon"
   950→              onClick={handleCleanup}
   951→              disabled={cleaning}
   952→              title="Cleanup expired files"
   953→            >
   954→              {cleaning ? (
   955→                <Loader2 className="h-4 w-4 animate-spin" />
   956→              ) : (
   957→                <RefreshCw className="h-4 w-4" />
   958→              )}
   959→            </Button>
   960→            <Button
   961→              variant="ghost"
   962→              size="icon"
   963→              onClick={() => setSettingsOpen(true)}
   964→              title="Settings"
   965→            >
   966→              <SettingsIcon className="h-4 w-4" />
   967→            </Button>
   968→          </div>
   969→        </div>
   970→      </header>
   971→
   972→      {/* Main */}
   973→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-6">
   974→        <div className="grid gap-6 lg:grid-cols-[360px_1fr]">
   975→          <QrPanel
   976→            sessions={sessions}
   977→            activeSession={activeSession}
   978→            onSelect={setActiveId}
   979→            onCreate={handleCreate}
   980→            onUpdate={handleUpdateSession}
   981→            onDelete={handleDeleteSession}
   982→            creating={creating}
   983→          />
   984→
   985→          {/* Files */}
   986→          <div className="flex flex-col gap-4">
   987→            <div className="flex flex-wrap items-center justify-between gap-2">
   988→              <div>
   989→                <h2 className="flex items-center gap-2 text-lg font-semibold">
   990→                  Received files
   991→                  {activeSession && (
   992→                    <Badge variant="secondary">{files.length}</Badge>
   993→                  )}
   994→                </h2>
   995→                <p className="text-xs text-muted-foreground">
   996→                  {activeSession
   997→                    ? activeSession.closed
   998→                      ? "Session closed — students can no longer upload."
   999→                      : "Updates live as students upload."
  1000→                    : "Select a session to view files."}
  1001→                </p>
  1002→              </div>
  1003→              <div className="flex items-center gap-2">
  1004→                <Button
  1005→                  variant="outline"
  1006→                  size="sm"
  1007→                  onClick={refreshFiles}
  1008→                  disabled={!activeId || loadingFiles}
  1009→                >
  1010→                  {loadingFiles ? (
  1011→                    <Loader2 className="mr-2 h-3.5 w-3.5 animate-spin" />
  1012→                  ) : (
  1013→                    <RefreshCw className="mr-2 h-3.5 w-3.5" />
  1014→                  )}
  1015→                  Refresh
  1016→                </Button>
  1017→                <Button
  1018→                  variant="outline"
  1019→                  size="sm"
  1020→                  onClick={() => setConfirmClearAll(true)}
  1021→                  disabled={files.length === 0}
  1022→                  className="text-destructive hover:bg-destructive hover:text-destructive-foreground"
  1023→                >
  1024→                  <Trash className="mr-2 h-3.5 w-3.5" />
  1025→                  Clear all
  1026→                </Button>
  1027→              </div>
  1028→            </div>
  1029→
  1030→            {!activeId ? (
  1031→              <div className="flex flex-col items-center justify-center gap-3 rounded-xl border border-dashed py-20 text-center">
  1032→                <FolderOpen className="h-12 w-12 text-muted-foreground/40" />
  1033→                <p className="text-sm text-muted-foreground">
  1034→                  No session selected.
  1035→                </p>
  1036→              </div>
  1037→            ) : files.length === 0 ? (
  1038→              <div className="flex flex-col items-center justify-center gap-3 rounded-xl border border-dashed py-20 text-center">
  1039→                <div className="flex h-16 w-16 items-center justify-center rounded-full bg-emerald-50 text-emerald-600">
  1040→                  <QrCode className="h-8 w-8" />
  1041→                </div>
  1042→                <div>
  1043→                  <p className="text-sm font-medium">Waiting for uploads</p>
  1044→                  <p className="text-xs text-muted-foreground">
  1045→                    Students scan the QR — files appear here instantly.
  1046→                  </p>
  1047→                </div>
  1048→              </div>
  1049→            ) : (
  1050→              <ScrollArea className="max-h-[calc(100vh-15rem)]">
  1051→                <div className="grid grid-cols-2 gap-3 pb-4 sm:grid-cols-3 xl:grid-cols-4">
  1052→                  {files.map((f) => (
  1053→                    <FileCard
  1054→                      key={f.id}
  1055→                      file={f}
  1056→                      onView={() => setViewFile(f)}
  1057→                      onPrint={() => handlePrint(f)}
  1058→                      onDelete={() => setConfirmDeleteFile(f)}
  1059→                      printing={printingId === f.id}
  1060→                    />
  1061→                  ))}
  1062→                </div>
  1063→              </ScrollArea>
  1064→            )}
  1065→          </div>
  1066→        </div>
  1067→      </main>
  1068→
  1069→      {/* Sticky footer */}
  1070→      <footer className="mt-auto border-t bg-background/90 backdrop-blur">
  1071→        <div className="mx-auto flex max-w-7xl flex-wrap items-center justify-between gap-3 px-4 py-3 text-xs">
  1072→          <div className="flex flex-wrap items-center gap-4 text-muted-foreground">
  1073→            <span>
  1074→              <b className="text-foreground">{files.length}</b> files
  1075→            </span>
  1076→            <span>
  1077→              <b className="text-foreground">{formatBytes(totalSize)}</b> total
  1078→            </span>
  1079→            <span>
  1080→              <b className="text-emerald-600">{printedCount}</b> printed
  1081→            </span>
  1082→            <span className="hidden sm:inline">
  1083→              {settings.autoDeleteAfterPrint
  1084→                ? "Auto-delete after print: on"
  1085→                : `Auto-expire: ${
  1086→                    settings.retentionHours > 0
  1087→                      ? settings.retentionHours + "h"
  1088→                      : "off"
  1089→                  }`}
  1090→            </span>
  1091→          </div>
  1092→          <span className="text-muted-foreground/70">
  1093→            QR File Collector · files stored locally &amp; auto-cleaned
  1094→          </span>
  1095→        </div>
  1096→      </footer>
  1097→
  1098→      <ViewFileDialog file={viewFile} onClose={() => setViewFile(null)} />
  1099→
  1100→      <SettingsDialog
  1101→        open={settingsOpen}
  1102→        onOpenChange={setSettingsOpen}
  1103→        settings={settings}
  1104→        onSave={handleSaveSettings}
  1105→      />
  1106→
  1107→      <AlertDialog open={confirmDeleteFile !== null} onOpenChange={(o) => !o && setConfirmDeleteFile(null)}>
  1108→        <AlertDialogContent>
  1109→          <AlertDialogHeader>
  1110→            <AlertDialogTitle>Delete this file?</AlertDialogTitle>
  1111→            <AlertDialogDescription>
  1112→              <b>{confirmDeleteFile?.filename}</b> by{" "}
  1113→              {confirmDeleteFile?.studentName || "Anonymous"} will be permanently
  1114→              removed from disk.
  1115→            </AlertDialogDescription>
  1116→          </AlertDialogHeader>
  1117→          <AlertDialogFooter>
  1118→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1119→            <AlertDialogAction
  1120→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
  1121→              disabled={deletingId !== null}
  1122→              onClick={() => {
  1123→                if (confirmDeleteFile) handleDeleteFile(confirmDeleteFile.id);
  1124→                setConfirmDeleteFile(null);
  1125→              }}
  1126→            >
  1127→              {deletingId && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
  1128→              Delete
  1129→            </AlertDialogAction>
  1130→          </AlertDialogFooter>
  1131→        </AlertDialogContent>
  1132→      </AlertDialog>
  1133→
  1134→      <AlertDialog open={confirmClearAll} onOpenChange={setConfirmClearAll}>
  1135→        <AlertDialogContent>
  1136→          <AlertDialogHeader>
  1137→            <AlertDialogTitle>Delete all files in this session?</AlertDialogTitle>
  1138→            <AlertDialogDescription>
  1139→              This removes {files.length} file(s) from disk. The session stays
  1140→              open for new uploads.
  1141→            </AlertDialogDescription>
  1142→          </AlertDialogHeader>
  1143→          <AlertDialogFooter>
  1144→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1145→            <AlertDialogAction
  1146→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
  1147→              onClick={handleClearAll}
  1148→            >
  1149→              Delete all
  1150→            </AlertDialogAction>
  1151→          </AlertDialogFooter>
  1152→        </AlertDialogContent>
  1153→      </AlertDialog>
  1154→    </div>
  1155→  );
  1156→}
