   162→   162→                {sessions.map((s) => (
   163→   163→                  <SelectItem key={s.id} value={s.id}>
   164→   164→                    <span className="flex items-center gap-2">
   165→   165→                      {s.closed && (
   166→   166→                        <Lock className="h-3 w-3 text-muted-foreground" />

   169→   169→                      <Badge
   170→   170→                        variant="secondary"
   171→   171→                        className="ml-1 h-4 px-1 text-[10px]"
   172→   172→                      >
   173→   173→                        {s.code}
   174→   174→                      </Badge>
   175→   175→                    </span>
   176→   176→                  </SelectItem>
   177→   177→                ))}
   178→   178→              </SelectContent>
   179→   179→            </Select>
   180→   180→            <Button
   181→   181→              size="icon"
   182→   182→              variant="outline"
   183→   183→              onClick={() => setShowNew(true)}
   184→   184→              title="New session"
   185→   185→            >
   186→   186→              <Plus className="h-4 w-4" />
   187→   187→            </Button>
   188→   188→          </div>
   189→   189→        </div>
   190→   190→
   191→   191→        {/* QR display */}
   192→   192→        {activeSession ? (
   193→   193→          <div className="flex flex-col items-center gap-3 rounded-xl border bg-gradient-to-b from-emerald-50/60 to-transparent p-4">
   194→   194→            <div className="rounded-xl bg-white p-3 shadow-sm ring-1 ring-black/5">
   195→   195→              <QRCodeSVG
   196→   196→                value={uploadUrl}
   197→   197→                size={208}
   198→   198→                level="M"
   199→   199→                marginSize={1}
   200→   200→                fgColor="#0f172a"
   201→   201→              />
   202→   202→            </div>
   203→   203→            <div className="flex w-full items-center justify-between rounded-lg bg-white px-3 py-2 ring-1 ring-black/5">
   204→   204→              <div className="min-w-0">
   205→   205→                <p className="text-[11px] uppercase tracking-wide text-muted-foreground">
   206→   206→                  Session code
   207→   207→                </p>
   208→   208→                <p className="font-mono text-lg font-bold tracking-widest text-slate-900">
   209→   209→                  {activeSession.code}
   210→   210→                </p>
   211→   211→              </div>
   212→   212→              <Button
   213→   213→                size="sm"
   214→   214→                variant="ghost"
   215→   215→                onClick={() => copy(activeSession.code, "code")}

   218→   218→                  <Check className="h-4 w-4 text-emerald-600" />

   220→   220→                  <Copy className="h-4 w-4" />
   221→   221→                )}
   222→   222→              </Button>
   223→   223→            </div>
   224→   224→
   225→   225→            <div className="flex w-full flex-col gap-2">
   226→   226→              <Button
   227→   227→                variant="outline"

   229→   229→                onClick={() => copy(uploadUrl, "link")}
   230→   230→              >
   231→   231→                {copied === "link" ? (
   232→   232→                  <Check className="h-4 w-4 text-emerald-600" />
   233→   233→                ) : (
   234→   234→                  <Copy className="h-4 w-4" />
   235→   235→                )}

   239→   239→                variant="outline"
   240→   240→                className="w-full justify-start"
   241→   241→                onClick={() => window.open(uploadUrl, "_blank")}
   242→   242→              >
   243→   243→                <ExternalLink className="h-4 w-4" />
   244→   244→                Open student page
   245→   245→              </Button>
   246→   246→            </div>
   247→   247→
   248→   248→            <Separator />
   249→   249→
   250→   250→            <div className="flex w-full items-center justify-between">
   251→   251→              <div className="flex items-center gap-2">
   252→   252→                <Label
   253→   253→                  htmlFor="closed-toggle"
   254→   254→                  className="text-sm text-muted-foreground"
   255→   255→                >
   256→   256→                  {activeSession.closed ? "Closed" : "Accepting uploads"}
   257→   257→                </Label>
   258→   258→              </div>

   280→   280→            <FolderOpen className="h-10 w-10 text-muted-foreground/50" />

   286→   286→              New session

   288→   288→          </div>
   289→   289→        )}

   291→   291→
   292→   292→      {/* New session dialog */}
   293→   293→      <Dialog open={showNew} onOpenChange={setShowNew}>
   294→   294→        <DialogContent>
   295→   295→          <DialogHeader>
   296→   296→            <DialogTitle>New collection session</DialogTitle>
   297→   297→            <DialogDescription>

   299→   299→            </DialogDescription>
   300→   300→          </DialogHeader>
   301→   301→          <div className="py-2">
   302→   302→            <Input
   303→   303→              autoFocus
   304→   304→              placeholder="Session name"
   305→   305→              value={newName}

   308→   308→                if (e.key === "Enter" && newName.trim() && !creating) {
   309→   309→                  onCreate(newName.trim());
   310→   310→                  setNewName("");
   311→   311→                  setShowNew(false);
   312→   312→                }
   313→   313→              }}
   314→   314→            />
   315→   315→          </div>
   316→   316→          <DialogFooter>
   317→   317→            <Button
   318→   318→              variant="outline"
   319→   319→              onClick={() => setShowNew(false)}
   320→   320→            >
   321→   321→              Cancel
   322→   322→            </Button>
   323→   323→            <Button
   324→   324→              disabled={!newName.trim() || creating}
   325→   325→              onClick={() => {
   326→   326→                onCreate(newName.trim());
   327→   327→                setNewName("");
   328→   328→                setShowNew(false);
   329→   329→              }}
   330→   330→            >
   331→   331→              {creating && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   332→   332→              Create
   333→   333→            </Button>
   334→   334→          </DialogFooter>
   335→   335→        </DialogContent>
   336→   336→      </Dialog>
   337→   337→
   338→   338→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   339→   339→        <AlertDialogContent>
   340→   340→          <AlertDialogHeader>
   341→   341→            <AlertDialogTitle>Delete this session?</AlertDialogTitle>

   343→   343→              This permanently deletes <b>{activeSession?.name}</b> and{" "}

   345→   345→            </AlertDialogDescription>
   346→   346→          </AlertDialogHeader>
   347→   347→          <AlertDialogFooter>
   348→   348→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   349→   349→            <AlertDialogAction
   350→   350→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   351→   351→              onClick={() => {
   352→   352→                if (activeSession) onDelete(activeSession.id);
   353→   353→                setConfirmDelete(false);
   354→   354→              }}
   355→   355→            >
   356→   356→              Delete
   357→   357→            </AlertDialogAction>
   358→   358→          </AlertDialogFooter>
   359→   359→        </AlertDialogContent>

   362→   362→  );
   363→   363→}

   370→   370→  file,
   371→   371→  onView,
   372→   372→  onPrint,
   373→   373→  onDelete,

   388→   388→        className="relative flex h-36 w-full items-center justify-center overflow-hidden bg-muted/40"

   450→   450→          <Button
   451→   451→            size="sm"

   454→   454→            onClick={onView}

   467→   467→            </a>

   473→   473→            onClick={onPrint}
   474→   474→            disabled={printing}
   475→   475→            title="Print"

   480→   480→              <Printer className="h-3.5 w-3.5" />
   481→   481→            )}
   482→   482→          </Button>

   484→   484→            size="sm"

   487→   487→            onClick={onDelete}
   488→   488→            title="Delete"
   489→   489→          >
   490→   490→            <Trash2 className="h-3.5 w-3.5" />
   491→   491→          </Button>
   492→   492→        </div>
   493→   493→      </CardContent>
   494→   494→    </Card>
   495→   495→  );
   496→   496→}

   503→   503→  file,
